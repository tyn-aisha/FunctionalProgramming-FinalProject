using System.Collections.Immutable;
using System.Diagnostics.CodeAnalysis;
using System.Globalization;
using IntergalacticJobCenter.Models;

namespace IntergalacticJobCenter.Input;

/// <summary>
/// Application boundary: converts raw CSV text into validated domain records.
/// No invalid value is ever replaced by a default.
/// Format: Name,Race,Tentacles,Eyes,Aggression,Compatibility (names must not contain commas).
/// </summary>
public static class ApplicantParser
{
    public const int FieldCount = 6;
    public const int MaxBodyParts = 64;   // upper limit for tentacles and eyes
    public const int MaxScale = 100;      // upper limit for aggression and compatibility

    /// <exception cref="ArgumentNullException">The dataset itself is missing (null).</exception>
    public static ParseOutcome ParseLines(IEnumerable<string?>? lines)
    {
        ArgumentNullException.ThrowIfNull(lines);

        var applicants = ImmutableArray.CreateBuilder<AlienApplicant>();
        var failures = ImmutableArray.CreateBuilder<ParseFailure>();
        var lineNumber = 0;
        var firstContentSeen = false;

        foreach (var line in lines)
        {
            lineNumber++;

            if (line is null)
            {
                failures.Add(new ParseFailure(lineNumber, "Line is missing (null)."));
                continue;
            }

            if (string.IsNullOrWhiteSpace(line))
            {
                continue; // blank separator lines are not records
            }

            if (!firstContentSeen)
            {
                firstContentSeen = true;
                if (IsHeader(line))
                {
                    continue;
                }
            }

            if (TryParseLine(line, out var applicant, out var error))
            {
                applicants.Add(applicant);
            }
            else
            {
                failures.Add(new ParseFailure(lineNumber, error));
            }
        }

        return new ParseOutcome(applicants.ToImmutable(), failures.ToImmutable());
    }

    public static bool TryParseLine(
        string? line,
        [NotNullWhen(true)] out AlienApplicant? applicant,
        out string error)
    {
        applicant = null;
        error = string.Empty;

        if (string.IsNullOrWhiteSpace(line))
        {
            error = "Line is empty.";
            return false;
        }

        var parts = line.Split(',');
        if (parts.Length != FieldCount)
        {
            error = $"Expected {FieldCount} comma-separated fields but found {parts.Length}.";
            return false;
        }

        var name = parts[0].Trim();
        if (name.Length == 0)
        {
            error = "Name is required.";
            return false;
        }

        var race = parts[1].Trim();
        if (race.Length == 0)
        {
            error = "Race is required.";
            return false;
        }

        if (!TryParseInt(parts[2], "Tentacles", 0, MaxBodyParts, out var tentacles, out error)) return false;
        if (!TryParseInt(parts[3], "Eyes", 0, MaxBodyParts, out var eyes, out error)) return false;
        if (!TryParseInt(parts[4], "Aggression", 0, MaxScale, out var aggression, out error)) return false;
        if (!TryParseInt(parts[5], "Compatibility", 0, MaxScale, out var compatibility, out error)) return false;

        applicant = new AlienApplicant(name, race, tentacles, eyes, aggression, compatibility);
        return true;
    }

    private static bool IsHeader(string line) =>
        line.TrimStart().StartsWith("name,", StringComparison.OrdinalIgnoreCase);

    private static bool TryParseInt(
        string text, string field, int min, int max, out int value, out string error)
    {
        error = string.Empty;

        if (string.IsNullOrWhiteSpace(text))
        {
            value = 0;
            error = $"{field} is missing.";
            return false;
        }

        var trimmed = text.Trim();
        if (!int.TryParse(trimmed, NumberStyles.Integer, CultureInfo.InvariantCulture, out value))
        {
            error = $"{field} '{trimmed}' is not a whole number.";
            return false;
        }

        if (value < min || value > max)
        {
            error = $"{field} {value} is outside the allowed range {min}-{max}.";
            return false;
        }

        return true;
    }
}
